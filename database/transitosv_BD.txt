CREATE DATABASE IF NOT EXISTS transitosv CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE transitosv;

-- 1. Tabla de Usuarios (Pasajeros y Administradores)
CREATE TABLE usuarios (
    id_usuario INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    correo VARCHAR(150) NOT NULL UNIQUE,
    contraseña VARCHAR(255) NOT NULL,
    rol VARCHAR(50) DEFAULT 'pasajero' -- Valores: pasajero, admin
) ENGINE=InnoDB;

-- 2. Tabla de Rutas de Transporte (ej. Ruta 30, Ruta 101-D, Ruta 44)
CREATE TABLE rutas (
    id_ruta INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL,
    lista_paradas TEXT NOT NULL -- Puede almacenar los puntos o JSON con las paradas clave
) ENGINE=InnoDB;

-- 3. Tabla de Unidades de Transporte (Buses y Microbuses con IoT)
CREATE TABLE unidades_transporte (
    id_unidad INT AUTO_INCREMENT PRIMARY KEY,
    coordenadas VARCHAR(100) NOT NULL, -- Lat, Lng actual en tiempo real
    velocidad DECIMAL(5,2) DEFAULT 0.00, -- Velocidad en km/h
    estado VARCHAR(50) NOT NULL, -- Valores: activo, en parada, inactivo
    id_ruta INT NOT NULL,
    CONSTRAINT fk_unidad_ruta FOREIGN KEY (id_ruta) REFERENCES rutas(id_ruta) ON DELETE CASCADE
) ENGINE=InnoDB;

-- 4. Tabla de Consultas (Historial de búsqueda de rutas y chat de IA por usuario)
CREATE TABLE consultas (
    id_consulta INT AUTO_INCREMENT PRIMARY KEY,
    origen VARCHAR(150) NOT NULL,
    destino VARCHAR(150) NOT NULL,
    tiempo_estimado INT NULL, -- Tiempo estimado en minutos devuelto por la IA
    id_usuario INT NOT NULL,
    CONSTRAINT fk_consulta_usuario FOREIGN KEY (id_usuario) REFERENCES usuarios(id_usuario) ON DELETE CASCADE
) ENGINE=InnoDB;

-- 5. Tabla de Telemetría (Historial de coordenadas enviadas por los dispositivos IoT)
CREATE TABLE telemetria (
    id_registro INT AUTO_INCREMENT PRIMARY KEY,
    fecha_hora DATETIME NOT NULL,
    coordenadas VARCHAR(100) NOT NULL,
    velocidad DECIMAL(5,2) DEFAULT 0.00,
    id_unidad INT NOT NULL,
    CONSTRAINT fk_telemetria_unidad FOREIGN KEY (id_unidad) REFERENCES unidades_transporte(id_unidad) ON DELETE CASCADE
) ENGINE=InnoDB;
